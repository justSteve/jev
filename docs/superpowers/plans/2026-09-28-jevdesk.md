> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `jevdesk`, a local decision-support agent that walks a profile-defined trading decision tree at five checkpoints and returns ACT / WAIT / PASS / NEED-INFO, using Jev only for single typed judgments.

**Bead:** `dr-nbv` (DReader beads; reference it in every commit message as `[dr-nbv]`)

**Spec:** `docs/superpowers/specs/2026-09-28-jevdesk-design.md` (read it first).

**Architecture:** A new package `jevdesk/` beside `jevloop/` in this repo. The profile (TOML) holds the tree: gate order, per-gate failure mode and thresholds, and every numeric absolute. `compute.py` is pure arithmetic. `questions.py` builds judged questions and runs each through jevloop's no-arithmetic guard. `judge.py` sends one checkpoint's questions to Jev in one call through jevloop's client. `engine.py` walks the gates and composes the verdict. `server.py` plus `web/index.html` give Steve a local form. `log.py` writes one JSON line per evaluation.

**Tech Stack:** Python ≥ 3.11 (stdlib `tomllib`, `http.server`, `dataclasses`), pytest, jevloop's `client.py` / `vault.py` / `split.py` / `battery.py`. No new dependencies.

- Tests: `.venv/bin/python -m pytest -q`. `tests/conftest.py` puts the repo root on `sys.path`, so `import jevdesk` works in tests.
- The baseline is 90 passing tests; that must stay green after every task.

```python
    for a computable quantity. Shared with jevdesk, which has its own
```

```python
"""Data shapes shared across jevdesk: gate results and verdicts."""

from __future__ import annotations

from dataclasses import asdict, dataclass, field
from enum import Enum


class Outcome(str, Enum):
    THROUGH = "through"
    PASS = "PASS"
    WAIT = "WAIT"
    NEED_INFO = "NEED-INFO"


@dataclass
class GateResult:
    gate: str
    outcome: Outcome
    reason: str
    source: str  # "computed", "fact" or "judged"
    advisory_failure: bool = False


@dataclass
class Verdict:
    checkpoint: int
    verdict: str  # "ACT", "WAIT", "PASS" or "NEED-INFO"
    deciding_gate: str | None
    gates: list[GateResult] = field(default_factory=list)
    plan: dict = field(default_factory=dict)
    missing: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)
    client: str = ""
    model: str = ""
    jev: dict = field(default_factory=dict)

    def to_dict(self) -> dict:
        d = asdict(self)
        d["gates"] = [{**asdict(g), "outcome": g.outcome.value} for g in self.gates]
        return d
```

```python
"""Pure arithmetic. Everything a number can decide is decided here, never by Jev."""

from __future__ import annotations


def fib_level(swing_low: float, swing_high: float, ratio: float, direction: str) -> float:
    """Retracement level. Long: measured down from the swing high (discount).
    Short: measured up from the swing low (premium)."""
    rng = swing_high - swing_low
    if direction == "long":
        return swing_high - ratio * rng
    return swing_low + ratio * rng


def zone_bounds(swing_low: float, swing_high: float, low: float, high: float,
                direction: str) -> tuple[float, float]:
    a = fib_level(swing_low, swing_high, low, direction)
    b = fib_level(swing_low, swing_high, high, direction)
    return (float(min(a, b)), float(max(a, b)))


def in_zone(price: float, bounds: tuple[float, float]) -> bool:
    return bounds[0] <= price <= bounds[1]


def gamma_regime(price: float, hvl: float, p_trans: float | None = None,
                 c_trans: float | None = None) -> str:
    """At or above the HVL is the positive side; between the HVL and the call
    transition level is positive transition. Below the HVL is negative; between
    the HVL and the put transition level is negative transition."""
    if price >= hvl:
        if c_trans is not None and price < c_trans:
            return "positive_transition"
        return "positive"
    if p_trans is not None and price > p_trans:
        return "negative_transition"
    return "negative"


def _side(bid: float, ask: float, ratio: float) -> str | None:
    if bid > 0 and bid >= ratio * ask:
        return "bid"
    if ask > 0 and ask >= ratio * bid:
        return "ask"
    return None


def stack_count(cells: list[list[float]], ratio: float) -> int:
    """Longest run of consecutive levels where the same side dominates by
    `ratio`. `cells` are diagonal [bid, ask] pairs, ordered by price."""
    best = run = 0
    prev = None
    for bid, ask in cells:
        side = _side(float(bid), float(ask), ratio)
        run = run + 1 if side is not None and side == prev else (1 if side else 0)
        prev = side
        best = max(best, run)
    return best


def delta_flipped(prev_delta: float, curr_delta: float, direction: str) -> bool:
    if direction == "long":
        return prev_delta < 0 < curr_delta
    return prev_delta > 0 > curr_delta


def stop_points(entry: float, stop: float) -> float:
    return abs(entry - stop)


def stop_on_correct_side(entry: float, stop: float, direction: str) -> bool:
    return stop < entry if direction == "long" else stop > entry


def position_size(risk_dollars: float, points: float, point_value: float) -> int:
    if points <= 0 or point_value <= 0:
        raise ValueError("stop distance and point value must be positive")
    return int(risk_dollars // (points * point_value))


def volatility_multiplier(atr: float | None, limit: float, fraction: float) -> float:
    if atr is None:
        return 1.0
    return fraction if atr > limit else 1.0


def parse_hhmm(text: str) -> int:
    """Minutes since midnight for 'HH:MM'; ValueError otherwise."""
    hh, sep, mm = text.partition(":")
    if sep != ":" or not (hh.isdigit() and mm.isdigit()) or len(mm) != 2:
        raise ValueError(f"not HH:MM: {text!r}")
    h, m = int(hh), int(mm)
    if not (0 <= h < 24 and 0 <= m < 60):
        raise ValueError(f"not HH:MM: {text!r}")
    return h * 60 + m


def before_cutoff(now: str, cutoff: str) -> bool:
    return parse_hhmm(now) < parse_hhmm(cutoff)


def open_points(entry: float, price: float, direction: str) -> float:
    return price - entry if direction == "long" else entry - price


def open_r(entry: float, stop: float, price: float, direction: str) -> float:
    return open_points(entry, price, direction) / stop_points(entry, stop)


def targets_beyond(entry: float, direction: str, targets: dict[str, float],
                   order: list[str]) -> list[list]:
    """Targets on the profitable side of entry, in the profile's order."""
    out = []
    for name in order:
        if name not in targets:
            continue
        price = float(targets[name])
        if (price > entry) if direction == "long" else (price < entry):
            out.append([name, price])
    return out
```

```python
class ProfileError(Exception):
    """The profile is missing a setting or holds an invalid one."""


@dataclass(frozen=True)
class GateSetting:
    on_fail: str
    threshold: float
    advisory: bool


class Profile:
    def __init__(self, data: dict, source: str):
        self.data = data
        self.source = source
        self.name = str(data.get("name") or Path(source).stem)

    def get(self, section: str, key: str):
        return self.data[section][key]

    def gate_order(self, checkpoint: int) -> list[str]:
        table = self.data.get("gates", {}).get(f"cp{checkpoint}", {})
        return list(table.get("order", CHECKPOINT_GATES[checkpoint]))

    def gate_setting(self, gate: str) -> GateSetting:
        g = self.data.get("gate", {}).get(gate, {})
        return GateSetting(
            on_fail=g.get("on_fail", "PASS"),
            threshold=float(g.get("threshold", self.data["thresholds"]["default"])),
            advisory=bool(g.get("advisory", False)),
        )

    def point_value(self, instrument: str) -> float | None:
        value = self.data["instruments"].get(instrument)
        return None if value is None else float(value)


def _fail(source: str, msg: str) -> None:
    raise ProfileError(f"{source}: {msg}")


def validate(data: dict, source: str) -> None:
    for section, keys in REQUIRED.items():
        if not isinstance(data.get(section), dict):
            _fail(source, f"missing section [{section}]")
        for key in keys:
            if key not in data[section]:
                _fail(source, f"missing [{section}] {key}")

    thresholds = [data["thresholds"]["default"]]
    thresholds += [g["threshold"] for g in data.get("gate", {}).values() if "threshold" in g]
    for t in thresholds:
        if not isinstance(t, (int, float)) or not 0.5 < t <= 1:
            _fail(source, f"threshold {t!r} must be in (0.5, 1]")

    z = data["zone"]
    if not 0 <= z["low"] < z["high"] <= 1:
        _fail(source, f"zone low/high {z['low']}/{z['high']} must satisfy 0 <= low < high <= 1")

    for cp_key, table in data.get("gates", {}).items():
        cp = int(cp_key[2:]) if cp_key.startswith("cp") and cp_key[2:].isdigit() else None
        if cp not in CHECKPOINT_GATES:
            _fail(source, f"[gates.{cp_key}] is not a gated checkpoint (1, 2, 3 or 5)")
        for gate in table.get("order", []):
            if gate not in CHECKPOINT_GATES[cp]:
                _fail(source, f"gate '{gate}' is not a checkpoint-{cp} gate")

    known = set(IN_TRADE_GATES).union(*CHECKPOINT_GATES.values())
    for gate, table in data.get("gate", {}).items():
        if gate not in known:
            _fail(source, f"[gate.{gate}] names an unknown gate")
        if table.get("on_fail", "PASS") not in ("PASS", "WAIT"):
            _fail(source, f"[gate.{gate}] on_fail must be PASS or WAIT")

    try:
        parse_hhmm(str(data["session"]["cutoff"]))
    except ValueError:
        _fail(source, f"[session] cutoff {data['session']['cutoff']!r} must be HH:MM")

    sp = data["session_path"]
    if not set(sp["pass_scenarios"]) <= set(sp["scenarios"]):
        _fail(source, "[session_path] pass_scenarios must all be listed in scenarios")

    m = data["management"]
    if m["breakeven_mode"] not in ("event", "points"):
        _fail(source, "[management] breakeven_mode must be 'event' or 'points'")
    for event, action in m["event_actions"].items():
        if event not in MANAGEMENT_EVENTS or action not in MANAGEMENT_ACTIONS:
            _fail(source, f"[management.event_actions] {event} = {action!r} is not allowed")

    for name, pv in data["instruments"].items():
        if not isinstance(pv, (int, float)) or pv <= 0:
            _fail(source, f"instrument {name} point value must be positive")


def load_profile(path: Path) -> Profile:
    path = Path(path)
    try:
        data = tomllib.loads(path.read_text(encoding="utf-8"))
    except (OSError, tomllib.TOMLDecodeError) as exc:
        raise ProfileError(f"{path}: {exc}") from exc
    validate(data, str(path))
    return Profile(data, str(path))
```

```python
"""Send one checkpoint's judged questions to Jev in a single call (speculative
fan-out). Any failure is JudgeUnavailable: the engine turns that into
NEED-INFO and never substitutes a default answer."""

from __future__ import annotations

import random

from jevloop.battery import REQUIRED_ANSWER_KEYS


class JudgeUnavailable(Exception):
    """Jev could not give a usable answer for this checkpoint."""


def ask(client, state: dict, questions: dict, timeout: float = 10.0) -> tuple[dict, dict]:
    meta = {"route": getattr(client, "name", ""), "model": getattr(client, "model", "")}
    if not questions:
        return {}, meta
    try:
        answers, meta = client.ask(state=state, questions=questions, timeout=timeout)
    except Exception as exc:  # any client failure means no judgment, never a guess
        raise JudgeUnavailable(f"{type(exc).__name__}: {exc}") from exc
    for qid, question in questions.items():
        answer = answers.get(qid)
        if not isinstance(answer, dict):
            raise JudgeUnavailable(f"no answer for '{qid}'")
        missing = REQUIRED_ANSWER_KEYS[question["type"]] - set(answer)
        if missing:
            raise JudgeUnavailable(f"answer '{qid}' is missing {sorted(missing)}")
    return answers, meta


def is_mock(client) -> bool:
    return str(getattr(client, "model", "")).startswith("mock-")


class ScriptedClient:
    """Fixed answers for tests and scenarios. Returns only the questions asked."""

    name = "SCRIPTED"
    model = "scripted"

    def __init__(self, answers: dict):
        self.answers = answers

    def ask(self, state: dict, questions: dict, timeout: float) -> tuple[dict, dict]:
        got = {q: self.answers[q] for q in questions if q in self.answers}
        return got, {"route": self.name, "model": self.model, "latency_ms": 0.0}


def _confidence(probs: dict) -> float:
    n = len(probs)
    peak = max(probs.values())
    return 1.0 if n <= 1 else max(0.0, min(1.0, (n * peak - 1) / (n - 1)))


class GenericMockClient:
    """Random but well-shaped answers to whatever is asked. Never real."""

    name = "MOCK"
    model = "mock-jevdesk-0.1"

    def __init__(self, seed: int | None = None):
        self._rng = random.Random(seed)

    def _weights(self, n: int) -> list[float]:
        w = [self._rng.random() + 1e-9 for _ in range(n)]
        total = sum(w)
        return [x / total for x in w]

    def ask(self, state: dict, questions: dict, timeout: float) -> tuple[dict, dict]:
        out = {}
        for qid, q in questions.items():
            if q["type"] == "noul":
                out[qid] = {"type": "noul", "noul": round(self._rng.uniform(0.1, 0.9), 4)}
            elif q["type"] == "choice":
                opts = list(q["criteria"])
                probs = dict(zip(opts, self._weights(len(opts))))
                out[qid] = {"type": "choice", "choice": max(probs, key=probs.get),
                            "probabilities": {k: round(v, 4) for k, v in probs.items()},
                            "confidence": round(_confidence(probs), 4)}
            else:
                probs = self._weights(len(q["criteria"]))
                keyed = {str(i): p for i, p in enumerate(probs)}
                out[qid] = {"type": "score", "score": round(sum(i * p for i, p in enumerate(probs)), 4),
                            "legend": {str(i): lvl for i, lvl in enumerate(q["criteria"])},
                            "probabilities": {k: round(v, 4) for k, v in keyed.items()},
                            "confidence": round(_confidence(keyed), 4)}
        return out, {"route": self.name, "model": self.model, "latency_ms": 0.0}


def resolve_client(mock: bool):
    """The real Jev client when a key is available; otherwise the generic mock,
    announced. jevloop's own mock answers jevloop's battery, not ours."""
    if mock:
        print("MOCK DECISION CLIENT (--mock): judgments are random, not real.")
        return GenericMockClient()
    from jevloop.client import MockDecisionClient, resolve_decision_client

    client = resolve_decision_client(mock=False)
    if isinstance(client, MockDecisionClient):
        print("jevdesk: no Jev key found, so using the generic mock. Verdicts are NOT real judgments.")
        return GenericMockClient()
    return client
```

```python
"""Walk a checkpoint's gates in the profile's order and compose the verdict.

Code owns every computed gate and the composition. Jev answers only the judged
gates, in one call per checkpoint. A fact Steve enters settles its gate without
asking Jev. The first gate that does not pass through decides the verdict.
"""

from __future__ import annotations

from dataclasses import dataclass, field

from . import compute
from .judge import JudgeUnavailable, ask, is_mock
from .model import GateResult, Outcome, Verdict
from .profile import Profile
from .questions import GATE_QUESTION, build


class InputError(ValueError):
    """The checkpoint's inputs are incomplete or inconsistent."""


REQUIRED_INPUTS: dict[int, tuple[str, ...]] = {
    1: ("swing_low", "swing_high", "price", "description"),
    2: ("direction", "swing_low", "swing_high", "price", "level_type", "time", "description"),
    3: ("model", "direction", "instrument", "entry", "stop", "description"),
    4: ("direction", "entry", "stop", "price", "description"),
    5: ("time", "consecutive_losses", "day_pnl", "self_report"),
}

# checkpoint-3 gates that belong to one entry model only
MODEL_GATES = {
    "aggression_failed": "reversal",
    "pressure_flipped": "reversal",
    "reading_matches": "reversal",
    "acceptance_confirmed": "continuation",
}


@dataclass
class Ctx:
    checkpoint: int
    inputs: dict
    profile: Profile
    computed: dict = field(default_factory=dict)
    answers: dict | None = None
    unavailable: str = ""
    questions: dict = field(default_factory=dict)
    meta: dict = field(default_factory=dict)


def _f(inputs: dict, key: str) -> float:
    try:
        return float(inputs[key])
    except (KeyError, TypeError, ValueError):
        raise InputError(f"{key} must be a number") from None


def _opt(inputs: dict, key: str) -> float | None:
    value = inputs.get(key)
    return None if value in (None, "") else _f(inputs, key)


def validate_inputs(checkpoint: int, inputs: dict, profile: Profile) -> None:
    if checkpoint not in REQUIRED_INPUTS:
        raise InputError(f"unknown checkpoint {checkpoint}")
    required = REQUIRED_INPUTS[checkpoint]
    missing = [k for k in required if inputs.get(k) in (None, "")]
    if missing:
        raise InputError("missing inputs: " + ", ".join(missing))
    if "direction" in required and inputs["direction"] not in ("long", "short"):
        raise InputError("direction must be 'long' or 'short'")
    if checkpoint in (1, 2) and _f(inputs, "swing_low") >= _f(inputs, "swing_high"):
        raise InputError("swing_low must be below swing_high")
    if checkpoint in (3, 4) and not compute.stop_on_correct_side(
            _f(inputs, "entry"), _f(inputs, "stop"), inputs["direction"]):
        raise InputError("stop is on the wrong side of entry")
    if checkpoint == 3:
        if inputs["model"] not in ("reversal", "continuation"):
            raise InputError("model must be 'reversal' or 'continuation'")
        if profile.point_value(str(inputs["instrument"])) is None:
            raise InputError(f"instrument '{inputs['instrument']}' has no point value in profile {profile.name}")
    if "time" in required:
        try:
            compute.parse_hhmm(str(inputs["time"]))
        except ValueError:
            raise InputError("time must be HH:MM") from None


# --- gate helpers ---------------------------------------------------------

def _through(gate: str, reason: str, source: str) -> GateResult:
    return GateResult(gate, Outcome.THROUGH, reason, source)


def _fail(gate: str, ctx: Ctx, reason: str, source: str) -> GateResult:
    return GateResult(gate, Outcome(ctx.profile.gate_setting(gate).on_fail), reason, source)


def _unavailable(gate: str, ctx: Ctx) -> GateResult:
    return GateResult(gate, Outcome.NEED_INFO, f"judgment unavailable ({ctx.unavailable})", "judged")


def _noul(gate: str, ctx: Ctx, qid: str) -> GateResult:
    if ctx.answers is None:
        return _unavailable(gate, ctx)
    p = float(ctx.answers[qid]["noul"])
    t = ctx.profile.gate_setting(gate).threshold
    if p >= t:
        return _through(gate, f"{qid}: yes ({p:.2f} >= {t:.2f})", "judged")
    if p <= 1 - t:
        return _fail(gate, ctx, f"{qid}: no ({p:.2f} <= {1 - t:.2f})", "judged")
    return GateResult(gate, Outcome.NEED_INFO,
                      f"{qid}: uncertain ({p:.2f}); describe {qid.replace('_', ' ')} in more detail",
                      "judged")


def _choice(gate: str, ctx: Ctx, qid: str, accepted) -> GateResult:
    if ctx.answers is None:
        return _unavailable(gate, ctx)
    answer = ctx.answers[qid]
    conf = float(answer["confidence"])
    t = ctx.profile.gate_setting(gate).threshold
    if conf < t:
        return GateResult(gate, Outcome.NEED_INFO,
                          f"{qid}: '{answer['choice']}' at confidence {conf:.2f} < {t:.2f}; describe it in more detail",
                          "judged")
    if answer["choice"] in accepted:
        return _through(gate, f"{qid}: {answer['choice']} ({conf:.2f})", "judged")
    return _fail(gate, ctx, f"{qid}: {answer['choice']} (needs one of {list(accepted)})", "judged")


# --- the gates ------------------------------------------------------------

def g_bias_clear(ctx: Ctx) -> GateResult:
    r = _choice("bias_clear", ctx, "bias", ("bullish", "bearish"))
    if r.outcome is Outcome.THROUGH:
        ctx.computed["bias"] = ctx.answers["bias"]["choice"]
    return r


def g_environment_tradeable(ctx: Ctx) -> GateResult:
    return _choice("environment_tradeable", ctx, "environment",
                   ctx.profile.get("environment", "tradeable"))


def g_session_path_ok(ctx: Ctx) -> GateResult:
    sp = ctx.profile.data["session_path"]
    accepted = [s for s in sp["scenarios"] if s not in sp["pass_scenarios"]]
    return _choice("session_path_ok", ctx, "session_path", accepted)


def g_level_allowed(ctx: Ctx) -> GateResult:
    level = str(ctx.inputs["level_type"])
    if level in ctx.profile.get("location", "allowed_types"):
        return _through("level_allowed", f"level type {level} is allowed", "computed")
    return _fail("level_allowed", ctx, f"level type {level} is not an allowed location", "computed")


def g_in_zone(ctx: Ctx) -> GateResult:
    i = ctx.inputs
    bounds = compute.zone_bounds(_f(i, "swing_low"), _f(i, "swing_high"),
                                 ctx.profile.get("zone", "low"), ctx.profile.get("zone", "high"),
                                 i["direction"])
    ctx.computed["zone"] = list(bounds)
    price = _f(i, "price")
    if compute.in_zone(price, bounds):
        return _through("in_zone", f"{price:g} is inside {bounds[0]:g}-{bounds[1]:g}", "computed")
    return _fail("in_zone", ctx, f"{price:g} is outside {bounds[0]:g}-{bounds[1]:g}", "computed")


def g_location_meaningful(ctx: Ctx) -> GateResult:
    if ctx.inputs["level_type"] not in ctx.profile.get("location", "judged_types"):
        return _through("location_meaningful", "objective level type; no judgment needed", "computed")
    return _noul("location_meaningful", ctx, "meaningful_location")


def g_before_cutoff(ctx: Ctx) -> GateResult:
    now, cutoff = str(ctx.inputs["time"]), ctx.profile.get("session", "cutoff")
    if compute.before_cutoff(now, cutoff):
        return _through("before_cutoff", f"{now} is before {cutoff}", "computed")
    return _fail("before_cutoff", ctx, f"{now} is at or after the {cutoff} cutoff", "computed")


def g_aggression_failed(ctx: Ctx) -> GateResult:
    return _noul("aggression_failed", ctx, "aggression_failed")


def _has_delta_fact(ctx: Ctx) -> bool:
    return _opt(ctx.inputs, "prev_delta") is not None and _opt(ctx.inputs, "curr_delta") is not None


def g_pressure_flipped(ctx: Ctx) -> GateResult:
    if _has_delta_fact(ctx):
        prev, curr = _f(ctx.inputs, "prev_delta"), _f(ctx.inputs, "curr_delta")
        if compute.delta_flipped(prev, curr, ctx.inputs["direction"]):
            return _through("pressure_flipped", f"delta flipped {prev:g} -> {curr:g}", "fact")
        return _fail("pressure_flipped", ctx, f"delta did not flip ({prev:g} -> {curr:g})", "fact")
    return _noul("pressure_flipped", ctx, "pressure_flipped")


def g_reading_matches(ctx: Ctx) -> GateResult:
    return _choice("reading_matches", ctx, "absorption_or_exhaustion",
                   ctx.profile.get("confirmation", "reversal_readings"))


def g_imbalance_confirmed(ctx: Ctx) -> GateResult:
    cells = ctx.inputs.get("cells")
    conf = ctx.profile.data["confirmation"]
    if not cells:
        if conf["imbalance_required"]:
            return GateResult("imbalance_confirmed", Outcome.NEED_INFO,
                              "enter the imbalance cells (diagonal bid/ask pairs)", "fact")
        return _through("imbalance_confirmed", "not required by this profile", "computed")
    n = compute.stack_count(cells, float(conf["imbalance_ratio"]))
    ctx.computed["stack"] = n
    if n >= int(conf["min_stack"]):
        return _through("imbalance_confirmed", f"stack of {n} at {conf['imbalance_ratio']}x", "fact")
    return _fail("imbalance_confirmed", ctx, f"stack of {n} < {conf['min_stack']}", "fact")


def g_acceptance_confirmed(ctx: Ctx) -> GateResult:
    return _noul("acceptance_confirmed", ctx, "acceptance_confirmed")


def g_stop_within_cap(ctx: Ctx) -> GateResult:
    pts = compute.stop_points(_f(ctx.inputs, "entry"), _f(ctx.inputs, "stop"))
    cap = float(ctx.profile.get("risk", "stop_cap_points"))
    ctx.computed["risk_points"] = pts
    if pts <= cap:
        return _through("stop_within_cap", f"stop {pts:g} points <= cap {cap:g}", "computed")
    return _fail("stop_within_cap", ctx, f"stop {pts:g} points > cap {cap:g}", "computed")


def g_size_positive(ctx: Ctx) -> GateResult:
    i, p = ctx.inputs, ctx.profile
    pv = p.point_value(str(i["instrument"]))
    pts = compute.stop_points(_f(i, "entry"), _f(i, "stop"))
    mult = compute.volatility_multiplier(_opt(i, "atr"), float(p.get("risk", "volatility_atr_limit")),
                                         float(p.get("risk", "volatility_fraction")))
    budget = float(p.get("account", "risk_per_trade")) * mult
    n = compute.position_size(budget, pts, pv)
    ctx.computed.update(contracts=n, risk_dollars=round(n * pts * pv, 2), size_multiplier=mult)
    if n >= 1:
        return _through("size_positive", f"{n} contracts, ${n * pts * pv:g} at risk", "computed")
    return _fail("size_positive", ctx,
                 f"risk budget ${budget:g} buys 0 contracts at {pts:g} points x ${pv:g}", "computed")


def g_loss_count_ok(ctx: Ctx) -> GateResult:
    n, cap = int(_f(ctx.inputs, "consecutive_losses")), int(ctx.profile.get("session", "max_consecutive_losses"))
    if n < cap:
        return _through("loss_count_ok", f"{n} consecutive losses < {cap}", "computed")
    return _fail("loss_count_ok", ctx, f"{n} consecutive losses reaches the limit of {cap}", "computed")


def g_daily_loss_ok(ctx: Ctx) -> GateResult:
    pnl, limit = _f(ctx.inputs, "day_pnl"), float(ctx.profile.get("session", "daily_loss_limit"))
    if pnl > -limit:
        return _through("daily_loss_ok", f"day P&L {pnl:g} is above -{limit:g}", "computed")
    return _fail("daily_loss_ok", ctx, f"day P&L {pnl:g} reaches the -{limit:g} limit", "computed")


def g_tilt_ok(ctx: Ctx) -> GateResult:
    if ctx.answers is None:
        return _unavailable("tilt_ok", ctx)
    answer = ctx.answers["tilt_level"]
    conf, t = float(answer["confidence"]), ctx.profile.gate_setting("tilt_ok").threshold
    if conf < t:
        return GateResult("tilt_ok", Outcome.NEED_INFO,
                          f"tilt_level at confidence {conf:.2f} < {t:.2f}; describe your state in more detail",
                          "judged")
    level, cap = float(answer["score"]), int(ctx.profile.get("tilt", "max_level"))
    if level < cap + 0.5:
        return _through("tilt_ok", f"tilt {level:.2f} within level {cap}", "judged")
    return _fail("tilt_ok", ctx, f"tilt {level:.2f} is past level {cap}", "judged")


GATES = {
    "bias_clear": g_bias_clear,
    "environment_tradeable": g_environment_tradeable,

    "loss_count_ok": g_loss_count_ok,
    "daily_loss_ok": g_daily_loss_ok,
    "tilt_ok": g_tilt_ok,
}


# --- composition ----------------------------------------------------------

def _applies(ctx: Ctx, gate: str) -> bool:
    model = MODEL_GATES.get(gate)
    return model is None or ctx.checkpoint != 3 or ctx.inputs["model"] == model


def _needs_judgment(ctx: Ctx, gate: str) -> bool:
    if gate not in GATE_QUESTION:
        return False
    if gate == "pressure_flipped":
        return not _has_delta_fact(ctx)
    if gate == "location_meaningful":
        return ctx.inputs["level_type"] in ctx.profile.get("location", "judged_types")
    return True


def _is_advisory(ctx: Ctx, gate: str) -> bool:
    if ctx.profile.gate_setting(gate).advisory:
        return True
    return gate == "pressure_flipped" and not ctx.profile.get("confirmation", "delta_flip_required")


def _judge(ctx: Ctx, client, qids: list[str], timeout: float) -> None:
    ctx.questions = build(ctx.profile, qids)
    state = {"checkpoint": ctx.checkpoint, "inputs": ctx.inputs, "computed": ctx.computed}
    try:
        ctx.answers, ctx.meta = ask(client, state, ctx.questions, timeout)
    except JudgeUnavailable as exc:
        ctx.answers, ctx.unavailable = None, str(exc)
        ctx.meta = {"route": getattr(client, "name", ""), "model": getattr(client, "model", "")}


def _finish(ctx: Ctx, v: Verdict, client) -> Verdict:
    v.client = str(ctx.meta.get("route", ""))
    v.model = str(ctx.meta.get("model", ""))
    v.jev = {"questions": ctx.questions, "answers": ctx.answers,
             "unavailable": ctx.unavailable or None, "mock": is_mock(client)}
    return v


def _plan(ctx: Ctx) -> dict:
    i, p, c = ctx.inputs, ctx.profile, ctx.computed
    if ctx.checkpoint == 1:
        plan = {"bias": c.get("bias"), "direction": None, "zone": None, "regime": c.get("regime")}
        if c.get("bias"):
            direction = "long" if c["bias"] == "bullish" else "short"
            plan["direction"] = direction
            plan["zone"] = list(compute.zone_bounds(_f(i, "swing_low"), _f(i, "swing_high"),
                                                    p.get("zone", "low"), p.get("zone", "high"), direction))
        return plan
    if ctx.checkpoint == 2:
        return {"zone": c.get("zone")}
    if ctx.checkpoint == 3:
        targets = {k: float(v) for k, v in (i.get("targets") or {}).items()}
        return {"direction": i["direction"], "entry": _f(i, "entry"), "stop": _f(i, "stop"),
                "risk_points": c.get("risk_points"), "contracts": c.get("contracts"),
                "risk_dollars": c.get("risk_dollars"),
                "targets": compute.targets_beyond(_f(i, "entry"), i["direction"], targets,
                                                  p.get("targets", "order"))}
    return {"status": "continue trading"}


def evaluate(checkpoint: int, inputs: dict, profile: Profile, client, timeout: float = 10.0) -> Verdict:
    validate_inputs(checkpoint, inputs, profile)
    ctx = Ctx(checkpoint, dict(inputs), profile)
    if checkpoint == 4:
        from .in_trade import evaluate_in_trade

        return _finish(ctx, evaluate_in_trade(ctx, client, timeout), client)

    if checkpoint == 1 and _opt(inputs, "hvl") is not None:
        ctx.computed["regime"] = compute.gamma_regime(
            _f(inputs, "price"), _f(inputs, "hvl"), _opt(inputs, "p_trans"), _opt(inputs, "c_trans"))

    gates = [g for g in profile.gate_order(checkpoint) if _applies(ctx, g)]
    _judge(ctx, client, [GATE_QUESTION[g] for g in gates if _needs_judgment(ctx, g)], timeout)

    v = Verdict(checkpoint=checkpoint, verdict="ACT", deciding_gate=None)
    for gate in gates:
        r = GATES[gate](ctx)
        if r.outcome in (Outcome.PASS, Outcome.WAIT) and _is_advisory(ctx, gate):
            r.advisory_failure = True
            v.warnings.append(f"{gate} ({r.outcome.value}, advisory): {r.reason}")
            r.outcome = Outcome.THROUGH
        v.gates.append(r)
        if r.outcome is not Outcome.THROUGH:
            v.verdict, v.deciding_gate = r.outcome.value, gate
            if r.outcome is Outcome.NEED_INFO:
                v.missing.append(r.reason)
            break
    if v.verdict == "ACT":
        v.plan = _plan(ctx)
    return _finish(ctx, v, client)
```

```python
"""Checkpoint 4: manage an open trade. Order of precedence: own side absorbed
again -> exit; breakeven by points (only if the profile allows it); otherwise
the management event Jev identifies, mapped to an action by the profile."""

from __future__ import annotations

from . import compute
from .model import GateResult, Outcome, Verdict


def evaluate_in_trade(ctx, client, timeout: float) -> Verdict:
    from .engine import _f, _judge

    i, p = ctx.inputs, ctx.profile
    entry, stop, price, direction = _f(i, "entry"), _f(i, "stop"), _f(i, "price"), i["direction"]
    pts = compute.open_points(entry, price, direction)
    r = round(compute.open_r(entry, stop, price, direction), 2)
    ctx.computed.update(open_points=pts, open_r=r)
    _judge(ctx, client, ["absorbed_again", "management_event"], timeout)
    v = Verdict(checkpoint=4, verdict="WAIT", deciding_gate=None)
    base = {"open_points": pts, "open_r": r}

    if ctx.answers is None:
        v.verdict, v.deciding_gate = "NEED-INFO", "management_event"
        v.missing.append(f"judgment unavailable ({ctx.unavailable})")
        return v

    t_abs = p.gate_setting("absorbed_again").threshold
    p_abs = float(ctx.answers["absorbed_again"]["noul"])
    v.gates.append(GateResult("absorbed_again", Outcome.THROUGH if p_abs < t_abs else Outcome.PASS,
                              f"absorbed_again: {p_abs:.2f} (threshold {t_abs:.2f})", "judged"))
    if p_abs >= t_abs:
        v.verdict, v.deciding_gate = "ACT", "absorbed_again"
        v.plan = {**base, "action": "exit", "reason": "own side absorbed a second time"}
        return v

    m = p.data["management"]
    if m["breakeven_mode"] == "points" and pts >= float(m["breakeven_points"]):
        v.verdict, v.deciding_gate = "ACT", "breakeven_points"
        v.plan = {**base, "action": "breakeven", "reason": f"open {pts:g} >= {m['breakeven_points']:g} points"}
        return v

    event = ctx.answers["management_event"]
    conf, t_ev = float(event["confidence"]), p.gate_setting("management_event").threshold
    if conf < t_ev:
        v.verdict, v.deciding_gate = "NEED-INFO", "management_event"
        v.missing.append(f"management_event '{event['choice']}' at confidence {conf:.2f} < {t_ev:.2f}; "
                         "describe what just happened in more detail")
        return v

    action = m["event_actions"].get(event["choice"], "hold")
    v.gates.append(GateResult("management_event", Outcome.THROUGH, f"event: {event['choice']} ({conf:.2f})", "judged"))
    v.deciding_gate = "management_event"
    v.plan = {**base, "action": action, "event": event["choice"]}
    v.verdict = "WAIT" if action == "hold" else "ACT"
    return v
```

```python
"""One JSON line per evaluation, and outcome lines appended against its id,
so Jev's probabilities can later be checked against what happened."""

from __future__ import annotations

import json
import time
import uuid
from pathlib import Path


def _append(path: Path, record: dict) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("a", encoding="utf-8") as f:
        f.write(json.dumps(record) + "\n")


def append_evaluation(path: Path, profile: str, checkpoint: int, inputs: dict, verdict: dict) -> str:
    eid = uuid.uuid4().hex[:12]
    _append(Path(path), {"kind": "evaluation", "id": eid, "ts": time.strftime("%Y-%m-%dT%H:%M:%S"),
                         "profile": profile, "checkpoint": checkpoint, "inputs": inputs, "verdict": verdict})
    return eid


def record_outcome(path: Path, eval_id: str, outcome: str, note: str = "") -> None:
    _append(Path(path), {"kind": "outcome", "id": eval_id, "ts": time.strftime("%Y-%m-%dT%H:%M:%S"),
                         "outcome": outcome, "note": note})


def read_log(path: Path) -> list[dict]:
    path = Path(path)
    if not path.exists():
        return []
    return [json.loads(line) for line in path.read_text(encoding="utf-8").splitlines() if line.strip()]
```

```python
"""Local HTTP server: the checkpoint page, and POST /evaluate. Binds to
127.0.0.1 only; the Jev key never leaves this process."""

from __future__ import annotations

import json
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path

from . import log
from .engine import InputError, evaluate
from .judge import is_mock

PAGE = Path(__file__).resolve().parent / "web" / "index.html"


def make_handler(profile, client, log_path: Path):
    class Handler(BaseHTTPRequestHandler):
        def _send(self, code: int, body: bytes, ctype: str) -> None:
            self.send_response(code)
            self.send_header("Content-Type", ctype)
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)

        def _json(self, code: int, obj) -> None:
            self._send(code, json.dumps(obj).encode(), "application/json")

        def do_GET(self) -> None:
            if self.path in ("/", "/index.html"):
                self._send(200, PAGE.read_bytes(), "text/html; charset=utf-8")
            elif self.path == "/profile":
                self._json(200, {"profile": profile.name, "client": getattr(client, "name", ""),
                                 "model": getattr(client, "model", ""), "mock": is_mock(client)})
            else:
                self._send(404, b"not found", "text/plain")

        def do_POST(self) -> None:
            if self.path != "/evaluate":
                self._send(404, b"not found", "text/plain")
                return
            try:
                length = int(self.headers.get("Content-Length", "0"))
                req = json.loads(self.rfile.read(length) or b"{}")
                checkpoint, inputs = int(req["checkpoint"]), dict(req["inputs"])
                verdict = evaluate(checkpoint, inputs, profile, client)
            except (InputError, KeyError, ValueError, TypeError) as exc:
                self._json(400, {"error": str(exc)})
                return
            d = verdict.to_dict()
            d["id"] = log.append_evaluation(log_path, profile.name, checkpoint, inputs, d)
            self._json(200, d)

        def log_message(self, fmt, *args) -> None:  # keep the terminal quiet
            pass

    return Handler


def serve(profile, client, log_path: Path, host: str = "127.0.0.1", port: int = 8766) -> None:
    httpd = ThreadingHTTPServer((host, port), make_handler(profile, client, log_path))
    tag = "MOCK" if is_mock(client) else getattr(client, "model", "")
    print(f"jevdesk: profile {profile.name} · client {getattr(client, 'name', '')} ({tag}) · http://{host}:{port}")
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        pass
    finally:
        httpd.server_close()
```
